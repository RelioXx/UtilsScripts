(() => {
    // Teardown previous instance if re-executed (handles both Pro and legacy instances)
    if (window.__transcriptonicPro) {
        try { window.__transcriptonicPro.stop(false); } catch (e) {}
        window.__transcriptonicPro = null;
    }
    if (window.__transcriptonicInstance) {
        try { window.__transcriptonicInstance.stop(false); } catch (e) {}
        window.__transcriptonicInstance = null;
    }
    document.getElementById("transcriptonic-console-widget")?.remove();
    document.getElementById("tt-pro-widget")?.remove();

    const SELECTORS = {
        USER_NAME: ".awLEm",
        MEETING_TITLE: ".u6vdEc",
        GOOGLE_SYMBOLS: ".google-symbols",
        TEXT_CALL_END: "call_end",
        TEXT_CAPTIONS_OFF: "closed_caption_off",
        TEXT_CAPTIONS_ON: "closed_caption",
        TEXT_CHAT: "chat",
        TRANSCRIPT_REGION: 'div[role="region"][tabindex="0"]',
        CHAT_LIVE_REGION: 'div[aria-live="polite"].Ge9Kpc'
    };

    const TIMEFORMAT = {
        year: "numeric",
        month: "2-digit",
        day: "2-digit",
        hour: "2-digit",
        minute: "2-digit",
        hour12: true
    };

    const YOU_IDENTIFIERS = ["you", "tú", "tu", "vous", "sie", "du", "você", "вы", "io", "u"];

    const mutationConfig = {
        childList: true,
        attributes: true,
        subtree: true,
        characterData: true
    };

    const STORAGE_KEY = "transcriptonic_pro_active_session";
    const HISTORY_KEY = "transcriptonic_pro_history";

    const state = {
        meetingSoftware: "Google Meet",
        userName: "You",
        transcript: [],
        chatMessages: [],
        stateTranscriptBlock: {
            timestamp: "",
            mutationTargetElement: null,
            personName: "",
            transcriptTextBuffer: ""
        },
        meetingStartTimestamp: new Date().toISOString(),
        meetingTitle: "Meeting",
        transcriptTargetNode: null,
        transcriptObserver: null,
        chatMessagesTargetNode: null,
        chatMessagesObserver: null,
        hasMeetingEnded: false,
        captionsHidden: false,
        monitorInterval: null,
        chatMonitorInterval: null,
        timerInterval: null,
        blinkInterval: null,
        uiElement: null,
        isCollapsed: false
    };

    // Auto-save to LocalStorage
    function saveToStorage() {
        try {
            const payload = {
                meetingSoftware: state.meetingSoftware,
                meetingTitle: state.meetingTitle,
                meetingStartTimestamp: state.meetingStartTimestamp,
                transcript: state.transcript,
                chatMessages: state.chatMessages,
                lastUpdated: new Date().toISOString()
            };
            localStorage.setItem(STORAGE_KEY, JSON.stringify(payload));
        } catch (e) {}
    }

    // Clean Title Detection (strips Meet prefixes/suffixes)
    function detectMeetingTitle() {
        const titleEl = document.querySelector(SELECTORS.MEETING_TITLE);
        if (titleEl && titleEl.innerText?.trim()) {
            state.meetingTitle = titleEl.innerText.trim();
            return;
        }

        let cleanTitle = document.title
            .replace(/^Meet\s*[-–—]\s*/i, "")
            .replace(/\s*[-–—]\s*Google Meet$/i, "")
            .replace(/\s*[-–—]\s*Meet$/i, "")
            .trim();

        if (cleanTitle && cleanTitle !== "Meet") {
            state.meetingTitle = cleanTitle;
            return;
        }

        const pathCode = window.location.pathname.replace(/^\//, "");
        if (pathCode) state.meetingTitle = `Meeting-${pathCode}`;
    }

    // User Name Detection
    function detectUserName() {
        const userEl = document.querySelector(SELECTORS.USER_NAME);
        if (userEl && userEl.textContent?.trim()) {
            state.userName = userEl.textContent.trim();
        }
    }

    // Push buffer to confirmed transcript (with international "You" replacement)
    function pushBufferToTranscript() {
        if (state.stateTranscriptBlock.personName && state.stateTranscriptBlock.transcriptTextBuffer) {
            const rawName = state.stateTranscriptBlock.personName.trim();
            const isLocalUser = YOU_IDENTIFIERS.includes(rawName.toLowerCase());
            const personName = isLocalUser ? state.userName : rawName;

            const cleanText = state.stateTranscriptBlock.transcriptTextBuffer.trim();
            if (cleanText) {
                state.transcript.push({
                    personName: personName,
                    timestamp: state.stateTranscriptBlock.timestamp || new Date().toISOString(),
                    transcriptText: cleanText
                });
                state.stateTranscriptBlock.personName = "";
                state.stateTranscriptBlock.transcriptTextBuffer = "";
                state.stateTranscriptBlock.mutationTargetElement = null;

                saveToStorage();
                updateUI();
            }
        }
    }

    // Speaker Name Resolver with fallback
    function resolveSpeakerName(mutationTargetElement) {
        let name = mutationTargetElement.previousSibling?.textContent?.trim();
        if (!name) {
            const parent = mutationTargetElement.parentElement;
            name = parent?.querySelector('[data-sender-name]')?.textContent?.trim()
                || parent?.firstElementChild?.textContent?.trim();
        }
        return name || "Speaker";
    }

    // Transcript Mutation Callback (TranscripTonic deduplication core)
    function transcriptMutationCallback(mutationsList) {
        mutationsList.forEach((mutation) => {
            try {
                if (mutation.type === "characterData") {
                    const mutationTargetElement = mutation.target.parentElement;
                    if (!mutationTargetElement) return;

                    const transcriptUIBlocks = [...mutationTargetElement.parentElement?.parentElement?.children || []];
                    // Meet leaves 2 non-transcript utility elements at the bottom; length - 3 is the active speaker block
                    const isLastButSecondElement = transcriptUIBlocks.length >= 3
                        ? transcriptUIBlocks[transcriptUIBlocks.length - 3] === mutationTargetElement.parentElement
                        : (transcriptUIBlocks.length > 0 && transcriptUIBlocks[transcriptUIBlocks.length - 1] === mutationTargetElement.parentElement);

                    if (isLastButSecondElement) {
                        const currentPersonName = resolveSpeakerName(mutationTargetElement);
                        const currentTranscriptText = mutationTargetElement.textContent;

                        if (currentPersonName && currentTranscriptText) {
                            if (!state.stateTranscriptBlock.mutationTargetElement) {
                                state.stateTranscriptBlock.mutationTargetElement = mutationTargetElement;
                                state.stateTranscriptBlock.personName = currentPersonName;
                                state.stateTranscriptBlock.timestamp = new Date().toISOString();
                                state.stateTranscriptBlock.transcriptTextBuffer = currentTranscriptText;
                            } else {
                                // Speaker changed: push previous speaker's block
                                if (state.stateTranscriptBlock.mutationTargetElement !== mutationTargetElement) {
                                    pushBufferToTranscript();

                                    state.stateTranscriptBlock.mutationTargetElement = mutationTargetElement;
                                    state.stateTranscriptBlock.personName = currentPersonName;
                                    state.stateTranscriptBlock.timestamp = new Date().toISOString();
                                    state.stateTranscriptBlock.transcriptTextBuffer = currentTranscriptText;
                                } else {
                                    // Same person speaking more; handle Google Meet's ~30min buffer wrap
                                    if ((currentTranscriptText.length - state.stateTranscriptBlock.transcriptTextBuffer.length) < -250) {
                                        pushBufferToTranscript();
                                        state.stateTranscriptBlock.mutationTargetElement = mutationTargetElement;
                                        state.stateTranscriptBlock.timestamp = new Date().toISOString();
                                    }
                                    state.stateTranscriptBlock.transcriptTextBuffer = currentTranscriptText;
                                }
                            }
                            updateUI();
                        } else {
                            // Speaker stopped and Meet cleared the block
                            pushBufferToTranscript();
                        }
                    }
                }
            } catch (err) {
                console.error("[TranscripTonic Pro] Caption parse error:", err);
            }
        });
    }

    // Resilient Container Finder (Supports multiple Meet layouts and languages)
    function findTranscriptContainer() {
        return document.querySelector(SELECTORS.TRANSCRIPT_REGION)
            || document.querySelector('div[jscontroller][aria-label*="caption" i]')
            || document.querySelector('div[jscontroller][aria-label*="subtítul" i]')
            || document.querySelector('div[jscontroller][aria-label*="sous-titre" i]')
            || document.querySelector('div[aria-live="assertive"][tabindex="0"]')
            || document.querySelector('.iTTPOb');
    }

    function startTranscriptMonitor() {
        state.transcriptTargetNode = null;

        function checkNode() {
            if (state.hasMeetingEnded) return;

            const activeNode = findTranscriptContainer();
            if (!activeNode) return;

            if (!state.transcriptTargetNode || activeNode !== state.transcriptTargetNode || !state.transcriptTargetNode.isConnected) {
                pushBufferToTranscript();

                if (state.transcriptObserver) {
                    state.transcriptObserver.disconnect();
                }

                state.transcriptTargetNode = activeNode;
                state.transcriptObserver = new MutationObserver(transcriptMutationCallback);
                state.transcriptObserver.observe(activeNode, mutationConfig);

                if (state.captionsHidden) {
                    applyCaptionVisibility(true);
                }

                console.log("%c[TranscripTonic Pro] Captions observer active.", "color: #2A9ACA; font-weight: bold;");
                updateUI();
            }
        }

        checkNode();
        state.monitorInterval = setInterval(checkNode, 1500);
    }

    // Stealth Mode: Hide / Show on-screen captions
    function applyCaptionVisibility(hidden) {
        state.captionsHidden = hidden;
        if (state.transcriptTargetNode) {
            state.transcriptTargetNode.style.opacity = hidden ? "0" : "1";
            state.transcriptTargetNode.style.pointerEvents = hidden ? "none" : "auto";
        }
        const btn = document.getElementById("tt-stealth-btn");
        if (btn) {
            btn.textContent = hidden ? "👁️ Show CC" : "🙈 Hide CC";
        }
    }

    function toggleCaptionVisibility() {
        applyCaptionVisibility(!state.captionsHidden);
    }

    // In-Call Chat Monitoring
    function chatMessagesMutationCallback() {
        try {
            const chatMessagesElement = document.querySelector(SELECTORS.CHAT_LIVE_REGION);
            if (chatMessagesElement && chatMessagesElement.children.length > 0) {
                const chatMessageElement = chatMessagesElement.lastChild?.firstChild?.firstChild?.lastChild;
                const personAndTimestampElement = chatMessageElement?.firstChild;
                const rawName = (personAndTimestampElement?.childNodes.length === 1)
                    ? state.userName
                    : personAndTimestampElement?.firstChild?.textContent?.trim();

                const isLocalUser = rawName && YOU_IDENTIFIERS.includes(rawName.toLowerCase());
                const personName = isLocalUser ? state.userName : (rawName || "Participant");
                const timestamp = new Date().toISOString();
                const chatMessageText = chatMessageElement?.lastChild?.lastChild?.firstChild?.firstChild?.firstChild?.textContent?.trim();

                if (personName && chatMessageText) {
                    const isExisting = state.chatMessages.some(m =>
                        m.personName === personName && m.chatMessageText === chatMessageText
                    );
                    if (!isExisting) {
                        state.chatMessages.push({ personName, timestamp, chatMessageText });
                        console.log(`%c[Chat] ${personName}: ${chatMessageText}`, "color: #34A853;");
                        saveToStorage();
                        updateUI();
                    }
                }
            }
        } catch (err) {}
    }

    function initChat() {
        function hookChatNode() {
            const chatNode = document.querySelector(SELECTORS.CHAT_LIVE_REGION);
            if (chatNode && chatNode !== state.chatMessagesTargetNode) {
                if (state.chatMessagesObserver) state.chatMessagesObserver.disconnect();
                state.chatMessagesTargetNode = chatNode;
                state.chatMessagesObserver = new MutationObserver(chatMessagesMutationCallback);
                state.chatMessagesObserver.observe(chatNode, mutationConfig);
                console.log("%c[TranscripTonic Pro] Chat messages hooked.", "color: #2A9ACA;");
            }
        }

        hookChatNode();
        state.chatMonitorInterval = setInterval(hookChatNode, 2500);
    }

    function ensureCaptionsTurnedOn() {
        const offIcons = Array.from(document.querySelectorAll(SELECTORS.GOOGLE_SYMBOLS))
            .filter(el => el.textContent?.trim() === SELECTORS.TEXT_CAPTIONS_OFF);

        if (offIcons.length > 0) {
            const btn = offIcons[0].closest("button") || offIcons[0];
            btn.click();
            console.log("%c[TranscripTonic Pro] Turned ON captions automatically.", "color: #2A9ACA; font-weight: bold;");
        } else {
            const onIcons = Array.from(document.querySelectorAll(SELECTORS.GOOGLE_SYMBOLS))
                .filter(el => el.textContent?.trim() === SELECTORS.TEXT_CAPTIONS_ON);
            if (onIcons.length === 0) {
                console.warn("%c[TranscripTonic Pro] Please make sure Captions (CC) are enabled in Google Meet.", "color: #FFA500; font-weight: bold;");
            }
        }
    }

    // Analytics Calculation
    function computeAnalytics() {
        const speakers = {};
        let totalWords = 0;

        state.transcript.forEach(block => {
            const words = block.transcriptText.split(/\s+/).filter(Boolean).length;
            totalWords += words;
            speakers[block.personName] = (speakers[block.personName] || 0) + words;
        });

        const speakerStats = Object.entries(speakers)
            .map(([name, words]) => ({
                name,
                words,
                percentage: totalWords ? Math.round((words / totalWords) * 100) : 0
            }))
            .sort((a, b) => b.words - a.words);

        return { totalWords, totalBlocks: state.transcript.length, speakerStats };
    }

    // Formatters: Text, Markdown, JSON
    function formatAsTxt() {
        let out = `${state.meetingTitle}\nDate: ${new Date(state.meetingStartTimestamp).toLocaleString()}\n\n`;
        out += `--------------- TRANSCRIPT ---------------\n\n`;
        state.transcript.forEach(b => {
            const time = new Date(b.timestamp).toLocaleString("default", TIMEFORMAT).toUpperCase();
            out += `${b.personName} (${time})\n${b.transcriptText}\n\n`;
        });
        if (state.chatMessages.length > 0) {
            out += `\n--------------- CHAT MESSAGES ---------------\n\n`;
            state.chatMessages.forEach(c => {
                const time = new Date(c.timestamp).toLocaleString("default", TIMEFORMAT).toUpperCase();
                out += `${c.personName} (${time})\n${c.chatMessageText}\n\n`;
            });
        }
        out += `\n---------------\nExported with TranscripTonic Pro\n---------------`;
        return out;
    }

    function formatAsMarkdown() {
        const { totalWords, speakerStats } = computeAnalytics();
        let md = `# Meeting Transcript: ${state.meetingTitle}\n\n`;
        md += `| Attribute | Details |\n|---|---|\n`;
        md += `| **Date & Time** | ${new Date(state.meetingStartTimestamp).toLocaleString()} |\n`;
        md += `| **Total Words** | ${totalWords.toLocaleString()} |\n`;
        md += `| **Dialogue Turns** | ${state.transcript.length} |\n\n`;

        if (speakerStats.length > 0) {
            md += `### 👥 Speaker Participation\n\n`;
            speakerStats.forEach(s => {
                md += `- **${s.name}**: ${s.words} words (${s.percentage}%)\n`;
            });
            md += `\n`;
        }

        md += `## 💬 Spoken Transcript\n\n`;
        state.transcript.forEach(b => {
            const time = new Date(b.timestamp).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
            md += `> **${b.personName}** *(${time})*\n>\n> ${b.transcriptText}\n\n`;
        });

        if (state.chatMessages.length > 0) {
            md += `## 📨 In-Call Chat\n\n`;
            state.chatMessages.forEach(c => {
                const time = new Date(c.timestamp).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
                md += `- **${c.personName}** (${time}): ${c.chatMessageText}\n`;
            });
            md += `\n`;
        }

        md += `---\n*Generated by TranscripTonic Pro*\n`;
        return md;
    }

    function formatAsJson() {
        return JSON.stringify({
            meetingSoftware: state.meetingSoftware,
            meetingTitle: state.meetingTitle,
            meetingStartTimestamp: state.meetingStartTimestamp,
            meetingEndTimestamp: state.hasMeetingEnded ? new Date().toISOString() : null,
            userName: state.userName,
            analytics: computeAnalytics(),
            transcript: state.transcript,
            chatMessages: state.chatMessages
        }, null, 2);
    }

    // Generic Download Trigger
    function downloadFile(content, extension, mimeType) {
        const invalidRegex = /[:?"*<>|~/\\\u{1}-\u{1f}\u{7f}\u{80}-\u{9f}\p{Cf}\p{Cn}]|^[.\u{0}\p{Zl}\p{Zp}\p{Zs}]|[.\u{0}\p{Zl}\p{Zp}\p{Zs}]$|^(CON|PRN|AUX|NUL|COM[1-9]|LPT[1-9])(?=\.|$)/gui;
        const title = (state.meetingTitle || "Meeting").replaceAll(invalidRegex, "_").trim();
        const timestamp = new Date(state.meetingStartTimestamp).toLocaleString("default", TIMEFORMAT).replace(/[\/:]/g, "-");
        const filename = `${title} (${timestamp}).${extension}`;

        const blob = new Blob([content], { type: mimeType });
        const url = URL.createObjectURL(blob);
        const a = document.createElement("a");
        a.href = url;
        a.download = filename;
        document.body.appendChild(a);
        a.click();
        setTimeout(() => {
            if (a.parentNode) a.remove();
            URL.revokeObjectURL(url);
        }, 1500);
        showToast(`Downloaded ${extension.toUpperCase()}`);
    }

    function triggerExport(format = "txt") {
        pushBufferToTranscript();
        if (state.transcript.length === 0 && state.chatMessages.length === 0) {
            alert("No speech or chat recorded yet. Please ensure Captions (CC) are enabled.");
            return;
        }

        if (format === "md") {
            downloadFile(formatAsMarkdown(), "md", "text/markdown;charset=utf-8");
        } else if (format === "json") {
            downloadFile(formatAsJson(), "json", "application/json;charset=utf-8");
        } else {
            downloadFile(formatAsTxt(), "txt", "text/plain;charset=utf-8");
        }
    }

    // Clipboard Copy with fallback for enterprise policies
    function copyToClipboard(text, label = "Transcript") {
        pushBufferToTranscript();
        if (navigator.clipboard && window.isSecureContext) {
            navigator.clipboard.writeText(text).then(() => {
                showToast(`${label} copied!`);
            }).catch(() => fallbackCopy(text, label));
        } else {
            fallbackCopy(text, label);
        }
    }

    function fallbackCopy(text, label) {
        try {
            const ta = document.createElement("textarea");
            ta.value = text;
            ta.style.position = "fixed";
            ta.style.left = "-9999px";
            ta.style.top = "0";
            document.body.appendChild(ta);
            ta.focus();
            ta.select();
            document.execCommand("copy");
            ta.remove();
            showToast(`${label} copied!`);
        } catch (e) {
            showToast("Clipboard copy blocked.");
            console.warn("[TranscripTonic Pro] Clipboard copy fallback failed:", e);
        }
    }

    function stopAndExport(autoDownload = true) {
        if (state.hasMeetingEnded) return;
        state.hasMeetingEnded = true;

        if (state.monitorInterval) clearInterval(state.monitorInterval);
        if (state.chatMonitorInterval) clearInterval(state.chatMonitorInterval);
        if (state.timerInterval) clearInterval(state.timerInterval);
        if (state.blinkInterval) clearInterval(state.blinkInterval);
        if (state.transcriptObserver) state.transcriptObserver.disconnect();
        if (state.chatMessagesObserver) state.chatMessagesObserver.disconnect();

        pushBufferToTranscript();
        saveToStorage();

        // Archive into last 10 meetings history
        try {
            let history = JSON.parse(localStorage.getItem(HISTORY_KEY) || "[]");
            history.push({
                meetingSoftware: state.meetingSoftware,
                meetingTitle: state.meetingTitle,
                meetingStartTimestamp: state.meetingStartTimestamp,
                meetingEndTimestamp: new Date().toISOString(),
                transcript: state.transcript,
                chatMessages: state.chatMessages
            });
            if (history.length > 10) history = history.slice(-10);
            localStorage.setItem(HISTORY_KEY, JSON.stringify(history));
        } catch (e) {}

        if (autoDownload && (state.transcript.length > 0 || state.chatMessages.length > 0)) {
            triggerExport("txt");
        }

        if (state.uiElement) {
            state.uiElement.remove();
            state.uiElement = null;
        }

        console.log("%c[TranscripTonic Pro] Session stopped.", "color: #2A9ACA; font-weight: bold;");
    }

    // Call End Detection with Event Delegation (immune to toolbar re-renders)
    function setupCallEndDetection() {
        document.body.addEventListener("click", (e) => {
            const rawTarget = e.target;
            const target = rawTarget instanceof Element ? rawTarget : rawTarget?.parentElement;
            if (!target) return;

            const endIcon = target.closest(`.${SELECTORS.GOOGLE_SYMBOLS.replace(".", "")}`);
            const isEndIcon = endIcon && endIcon.textContent?.trim() === SELECTORS.TEXT_CALL_END;
            const endBtn = target.closest('button[data-tooltip*="leave" i], button[aria-label*="leave" i], button[aria-label*="salir" i]')
                || (isEndIcon ? (target.closest("button") || target) : null);

            if (endBtn) {
                console.log("%c[TranscripTonic Pro] Leave call clicked. Auto-exporting...", "color: #2A9ACA; font-weight: bold;");
                stopAndExport(true);
            }
        }, true);

        window.addEventListener("beforeunload", () => {
            pushBufferToTranscript();
            saveToStorage();
        });

        window.addEventListener("pagehide", () => {
            pushBufferToTranscript();
            saveToStorage();
        });
    }

    function makeDraggable(el) {
        let isDragging = false, startX, startY, origX, origY;
        el.addEventListener("mousedown", (e) => {
            const rawTarget = e.target;
            const target = rawTarget instanceof Element ? rawTarget : rawTarget?.parentElement;
            if (!target || ["BUTTON", "INPUT", "TEXTAREA"].includes(target.tagName)) return;
            isDragging = true;
            startX = e.clientX; startY = e.clientY;
            const rect = el.getBoundingClientRect();
            origX = rect.left; origY = rect.top;
            el.style.bottom = "auto";
            el.style.left = `${origX}px`;
            el.style.top = `${origY}px`;

            const onMouseMove = (ev) => {
                if (!isDragging) return;
                el.style.left = `${Math.max(10, Math.min(window.innerWidth - el.offsetWidth - 10, origX + (ev.clientX - startX)))}px`;
                el.style.top = `${Math.max(10, Math.min(window.innerHeight - el.offsetHeight - 10, origY + (ev.clientY - startY)))}px`;
            };
            const onMouseUp = () => {
                isDragging = false;
                document.removeEventListener("mousemove", onMouseMove);
                document.removeEventListener("mouseup", onMouseUp);
            };
            document.addEventListener("mousemove", onMouseMove);
            document.addEventListener("mouseup", onMouseUp);
        });
    }
